/** Verifier probe: following purchase-list cursors must visit every purchase exactly once (ADR 0011 exclusive cursors). */
@SpringBootTest
class PurchaseListCursorProbeIntegrationTest {
    @DynamicPropertySource
    static void configure(DynamicPropertyRegistry registry) throws SQLException {
        TenantSecurityIntegrationFixture.configure(registry);
        registry.add("spring.datasource.hikari.maximum-pool-size", () -> "4");
    }

    @Autowired PurchaseService purchases;
    @Autowired CustomerService customers;
    @Autowired TenantRepository tenants;
    @Autowired TenantMembershipRepository memberships;
    @Autowired TenantContextProvider contexts;
    @Autowired AuditExecutionContext auditExecution;

    @Test
    void followingCursorsVisitsEveryPurchaseExactlyOnce() throws Exception {
        Instant now = Instant.now();
        Tenant tenant = seedTenant(tenants, contexts, "Probe " + UUID.randomUUID(), now);
        TenantContext ctx = context(seedMembership(memberships, contexts, tenant.id(),
                new IdentityId(UUID.randomUUID()), TenantRole.OWNER, now));
        Customer customer = inContext(ctx, () -> customers.create("Probe customer", null,
                List.of(new PhoneInput("8888" + String.format("%04d", Math.abs(UUID.randomUUID().hashCode()) % 10000), "CR", true))));
        List<PurchaseId> expected = new ArrayList<>();
        for (int i = 0; i < 5; i++) {
            Instant at = now.minusSeconds(60L * (i + 1));
            expected.add(inContext(ctx, () -> purchases.record(customer.id(), at, "Item", "probe-" + UUID.randomUUID())).purchase().id());
        }
        for (int limit : new int[] {1, 2, 4}) {
            List<PurchaseId> seen = new ArrayList<>();
            PurchaseCursor cursor = null;
            for (int guard = 0; guard < 10; guard++) {
                PurchaseCursor before = cursor;
                PurchasePage page = inContext(ctx, () -> purchases.list(customer.id(), null, before, limit));
                page.purchases().forEach(p -> seen.add(p.id()));
                System.out.println("PROBE limit=" + limit + " page=" + page.purchases().stream().map(p -> expected.indexOf(p.id())).toList()
                        + " next=" + (page.nextCursor() != null));
                if (page.nextCursor() == null) break;
                cursor = PurchaseCursor.decode(page.nextCursor());
            }
            assertThat(seen).as("limit=" + limit).containsExactlyElementsOf(expected);
        }
    }

    private <T> T inContext(TenantContext context, Callable<T> operation) throws Exception {
        return auditExecution.callWithCorrelation(UUID.randomUUID(),
                () -> contexts.callWithContext(context, operation::call));
    }
}

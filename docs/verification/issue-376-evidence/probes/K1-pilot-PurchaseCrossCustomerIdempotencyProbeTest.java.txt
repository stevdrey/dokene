/** Probe: ADR 0011 says a key replay matches only with the same customer, timestamp and description. */
@SpringBootTest
class PurchaseCrossCustomerIdempotencyProbeTest {
    @DynamicPropertySource
    static void configure(DynamicPropertyRegistry registry) throws SQLException {
        TenantSecurityIntegrationFixture.configure(registry);
        registry.add("spring.datasource.hikari.maximum-pool-size", () -> "4");
    }

    @Autowired PurchaseService purchases;
    @Autowired CustomerService customers;
    @Autowired TenantRepository tenants;
    @Autowired TenantMembershipRepository memberships;
    @Autowired TenantContextProvider contexts;
    @Autowired AuditExecutionContext auditExecution;

    @Test
    void sameKeyOnAnotherCustomerOfSameTenantIsAConflict() throws Exception {
        Instant now = Instant.now();
        Tenant tenant = seedTenant(tenants, contexts, "Probe " + UUID.randomUUID(), now);
        TenantContext ctx = context(seedMembership(memberships, contexts, tenant.id(),
                new IdentityId(UUID.randomUUID()), TenantRole.OWNER, now));
        Customer a = inContext(ctx, () -> customers.create("Probe A", null, List.of(new PhoneInput(phone(), "CR", true))));
        Customer b = inContext(ctx, () -> customers.create("Probe B", null, List.of(new PhoneInput(phone(), "CR", true))));
        Instant at = Instant.now().minusSeconds(10);
        var created = inContext(ctx, () -> purchases.record(a.id(), at, "Same", "cross-customer-key"));
        assertThatThrownBy(() -> {
            var replay = inContext(ctx, () -> purchases.record(b.id(), at, "Same", "cross-customer-key"));
            System.out.println("PROBE replay on customer B returned purchase " + replay.purchase().id().value()
                    + " owned by customer " + replay.purchase().customerId().value()
                    + " (A=" + a.id().value() + ", B=" + b.id().value() + ", created=" + replay.created()
                    + ", sameAsA=" + replay.purchase().id().equals(created.purchase().id()) + ")");
        }).isInstanceOf(PurchaseConflictException.class);
    }

    private static String phone() {
        return "8888" + String.format("%04d", Math.abs(UUID.randomUUID().hashCode()) % 10000);
    }

    private <T> T inContext(TenantContext context, Callable<T> operation) throws Exception {
        return auditExecution.callWithCorrelation(UUID.randomUUID(),
                () -> contexts.callWithContext(context, operation::call));
    }
}
